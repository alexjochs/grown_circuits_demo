cmake_minimum_required(VERSION 3.20)
project(freerun LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(FREERUN_LIBDIR ${CMAKE_SOURCE_DIR}/python/freerun/_lib CACHE PATH "where the shared libraries are placed")

find_package(Threads REQUIRED)
add_library(freerun_cpu SHARED csrc/cpu_engine.cpp)
target_include_directories(freerun_cpu PRIVATE csrc)
target_compile_options(freerun_cpu PRIVATE -Wall -Wextra)
target_link_libraries(freerun_cpu PRIVATE Threads::Threads)
set_target_properties(freerun_cpu PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${FREERUN_LIBDIR})

option(FREERUN_CUDA "build the CUDA backend" OFF)
if(FREERUN_CUDA)
  enable_language(CUDA)
  if(NOT CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 90)
  endif()
  add_library(freerun_cuda SHARED csrc/cuda_engine.cu)
  target_compile_options(freerun_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  target_include_directories(freerun_cuda PRIVATE csrc)
  set_target_properties(freerun_cuda PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${FREERUN_LIBDIR})
endif()
